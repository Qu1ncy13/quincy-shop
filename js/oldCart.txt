function checkEmptyCart(){
    const items = document.querySelectorAll(".cart-item");
    if (cart.length === 0){
        emptyCart.classList.remove("hidden");
        cartItems.classList.add("hidden");
    }
    else{
        emptyCart.classList.add("hidden");
        cartItems.classList.remove("hidden");
    }
}

function updateCartTotal(){
    const allItemTotal = document.querySelectorAll(".item-total");
    let sum = 0;
    allItemTotal.forEach((item) => {
        let itemValue = Number(item.textContent.replace("$ ", ""));
        sum+=itemValue;
    });
    totalSum.textContent = "$ " + sum.toFixed(2);
}

const plusButton = document.querySelectorAll(".count-btn-plus");
const minusButton = document.querySelectorAll(".count-btn-minus");
plusButton.forEach((button) => {
    button.addEventListener("click",() => {
        const input = button.parentElement.querySelector("input");
        const priceItem = button.parentElement.parentElement.querySelector(".item-total");
        const basePrice = Number(priceItem.dataset.price);
        
        let inputValue = Number(input.value);
        if (inputValue < 99){
            inputValue++;
            input.value = inputValue;
        };
        priceItem.textContent = "$ " + (basePrice * inputValue).toFixed(2);
        updateCartTotal();
    });
});
minusButton.forEach((button) => {
    button.addEventListener("click", ()=> {

        const input = button.parentElement.querySelector("input");
        const priceItem = button.parentElement.parentElement.querySelector(".item-total");
        const basePrice = Number(priceItem.dataset.price);

        let inputValue = Number(input.value);
        if (inputValue > 1){
            inputValue--;
            input.value = inputValue;
        };
        priceItem.textContent = "$ " + (basePrice * inputValue).toFixed(2);
        updateCartTotal();
    });
});
const deleteItemBtn = document.querySelectorAll(".deleteBtn")
deleteItemBtn.forEach((button)=> {
    button.addEventListener("click", ()=>{
        const item = button.parentElement;
        item.remove();
        updateCartTotal();
        checkEmptyCart();
    });
});

const cleanCartConfirmBtn = document.querySelector(".confirm-btn");
const cleanCartCancelBtn = document.querySelector(".cancel-btn");
cleanCartConfirmBtn.addEventListener("click", ()=>{
    const cartEl = document.querySelectorAll(".cart-item");
    cartEl.forEach((element)=>{
        element.remove();
        emptyCart.classList.remove("hidden");
        cleanCartButton.classList.add("hidden");
        cartItems.classList.add("hidden");
        updateCartTotal();
    });
    cartModal.classList.add("hidden");
});

cleanCartCancelBtn.addEventListener("click", ()=>{
    cartModal.classList.add("hidden");
});
